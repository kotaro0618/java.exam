問７
SELECT prefectures.name AS prefecture,genders.name AS gender,SUM(populations.population) AS population
FROM populations
JOIN prefectures
ON prefectures.code=populations.prefecture_code
JOIN genders
ON genders.code=populations.gender_code
GROUP BY prefecture,genders.name,prefectures.code
HAVING prefectures.name IN('福岡県','佐賀県','長崎県','熊本県','大分県','宮崎県','鹿児島県','沖縄県')
ORDER BY prefectures.code,prefecture,genders.name DESC;
